# ONTAP Cluster Deployment Playbook (CLI, copy/paste)
# Repo    : netapp-ontap-powershell
# Target  : NetApp AFF / ASA (A70-class), ONTAP 9.13+
# Scope   : Single cluster, Day-1 build, peer-capable (intercluster LIFs;
#           cluster peer NOT formed)
# Author  : HumbledGeeks / Allen Johnson
# Version : 1.0
###############################################################################
#
# WHAT THIS PERFORMS
# ------------------
# This playbook walks a two-node ONTAP cluster from first console access
# through accepted handoff. It produces:
#   * Hardened cluster baseline (NTP, DNS, banner)
#   * Security baseline (OKM, AD-integrated admin auth + break-glass local,
#     insecure protocols disabled, hardened SSH ciphers)
#   * Network layer 2 with LACP, VLAN tagging, MTU 9000 on storage BCs,
#     failover groups per protocol
#   * Aggregates (ADPv2 verified) and HA enabled
#   * Two SVMs (names are configurable via <NAS_SVM> and <SAN_SVM>):
#       - <NAS_SVM> : NFSv3 + NFSv4.1 and SMB/CIFS
#                     (AD-joined, signing + encryption, ABE)
#       - <SAN_SVM> : iSCSI (CHAP) and FCP
#     (NAS and SAN are intentionally separated per repo CLAUDE.md.)
#   * Data LIFs per SVM/protocol with correct service-policies
#   * Intercluster LIFs ready for future SnapMirror peering
#   * Acceptance tests (HA takeover/giveback, LIF failover, snapshot, ASUP)
#   * Monitoring wiring (AutoSupport HTTPS, AIQUM, syslog, SNMPv3, EMS)
#   * Handoff evidence (config backup + show output bundle)
#
# OUT OF SCOPE: SnapMirror policy/relationship config, host-side mount/
# connect, application onboarding.
#
# HOW TO USE
# ----------
# 1. Open this file alongside your site worksheet (IPs, VLANs, node names,
#    AD domain, etc.).
# 2. Do a find-and-replace on every <PLACEHOLDER> below for your site.
#    Placeholders use UPPER_SNAKE_CASE inside angle brackets.
# 3. Work top-to-bottom, one phase at a time. Stop at each "# VERIFY:" line
#    and confirm output before proceeding.
# 4. NEVER paste secrets into this file. Passphrases and AD credentials are
#    shown as <PROMPTED_AT_RUNTIME> and entered at the cluster's prompt.
# 5. Lines starting with "#" are comments. Lines starting with "##" are
#    section dividers. Lines starting with "###" mark a new phase.
#
# PLACEHOLDER LEGEND (most common)
# --------------------------------
#   <CLUSTER_NAME>            cluster shortname (e.g. cluster)
#   <CLUSTER_FQDN>            cluster mgmt FQDN (e.g. example-cluster.example.com)
#   <CLUSTER_MGMT_IP/MASK/GW> cluster mgmt LIF address
#   <NODE1_NAME> <NODE2_NAME> node shortnames (e.g. example-cluster-01/-02)
#   <NODE1_MGMT_IP> etc.      node mgmt LIF addresses
#   <SP1_IP> <SP2_IP>         Service Processor addresses
#   <MGMT_VLAN_ID>            cluster mgmt VLAN tag
#   <ICL_VLAN_ID>             intercluster VLAN tag (separate from mgmt)
#   <NAS_VLAN_ID>             NFS+SMB data VLAN tag
#   <SAN_VLAN_ID>             iSCSI data VLAN tag
#   <PORT_A> <PORT_B> ...     physical data ports (e.g. e2a, e2b, e2c, e2d)
#   <NAS_SVM>                 SVM serving NFS + CIFS    (suggested name: svm-nas)
#   <SAN_SVM>                 SVM serving iSCSI + FCP   (suggested name: svm-san)
#   <AD_DOMAIN>               AD DNS domain (e.g. example.com)
#   <AD_OU>                   OU DN for SMB server computer object
#   <AD_ADMIN_GROUP>          AD group mapped to cluster admin role
#   <AD_RO_GROUP>             AD group mapped to readonly role
#   <SYSLOG_SERVER>           syslog/SIEM target
#   <NTP_SERVER_1/2>          NTP sources
#   <DNS_SERVER_1/2>          DNS resolvers
#   <CIDR_CLIENTS>            allowed NFS client CIDR (NOT 0.0.0.0/0)
#
# NOTE on LIF names: data-LIF names follow a fixed convention
#   n<N>_<proto>_lif<NN>      e.g. n1_nfs_lif01, n2_cifs_lif02
# so they are not parameterized -- each LIF has a unique, self-describing
# name. If you change <NAS_SVM>/<SAN_SVM>, the LIF names do NOT change.
#
###############################################################################


### Phase 0 -- Pre-deployment checklist (off-array; no CLI required)
##############################################################################
# Complete BEFORE the cluster setup wizard. No commands here; verify each
# item and check it off.
#
#   [ ] Hardware racked, powered, cabled per topology drawing
#   [ ] All disks present and same firmware level
#   [ ] Site worksheet complete (cluster + node + LIF IPs, VLANs, DNS, NTP)
#   [ ] DNS A/PTR records pre-created for <CLUSTER_FQDN>, <NODE*_FQDN>
#   [ ] AD service account ready (rights to join SMB server in <AD_OU>)

#   [ ] CA available to sign the cluster mgmt cert
#   [ ] Syslog/SIEM endpoint reachable from cluster mgmt

##############################################################################

### Phase 2 -- Security baseline
##############################################################################
### Onboard Key Manager (OKM) ###
# Passphrase is entered at the prompt -- do NOT paste it into this file.
# Store the passphrase in your enterprise secret vault immediately.
security key-manager onboard enable
#   Passphrase: <PROMPTED_AT_RUNTIME>
#   Re-enter:   <PROMPTED_AT_RUNTIME>

# VERIFY:
security key-manager onboard show-backup     # save backup to secret vault
security key-manager key query                # auth keys present
#
# ALTERNATIVE (recommended for regulated production): external KMIP
#   security key-manager external enable -vserver <CLUSTER_NAME> \
#       -key-servers <KMIP_HOST_1>:5696,<KMIP_HOST_2>:5696 \
#       -client-cert <CLIENT_CERT_NAME> -server-ca-certs <CA_CERT_NAME>

### Disable insecure protocols / harden SSH ###
security protocol modify -application telnet -enabled false
security protocol modify -application rsh    -enabled false
security ssh modify -vserver <CLUSTER_NAME> \
    -ciphers aes256-gcm@openssh.com,aes128-gcm@openssh.com,aes256-ctr,aes128-ctr \
    -key-exchange-algorithms diffie-hellman-group-exchange-sha256,curve25519-sha256 \
    -mac-algorithms hmac-sha2-512,hmac-sha2-256
# Disable SNMPv1/v2c (we configure SNMPv3 in Phase 11)
system snmp community delete -community-name public      # if present
system snmp community delete -community-name private     # if present

# VERIFY:
security protocol show
security ssh show -vserver <CLUSTER_NAME>
system snmp community show

### Break-glass local admin (long random password; store in secret vault) ###
security login create -user-or-group-name breakglass -application ssh \
    -authentication-method password -role admin
#   Password: <PROMPTED_AT_RUNTIME>
security login password-policy modify -vserver <CLUSTER_NAME> \
    -minimum-password-length 16 -minimum-symbol-characters 1 \
    -minimum-digit-characters 1 -minimum-uppercase-characters 1 \
    -minimum-lowercase-characters 1 -minimum-age 1
# Delete the default 'diag' password if it's set / disable unused accounts:
security login show
#   Lock or remove any unused 'admin'-named local accounts other than 'admin' itself.

# VERIFY end of Phase 2:
security login show -vserver <CLUSTER_NAME>
security key-manager onboard show

### Phase 4 -- Network layer 2 (broadcast domains, ifgrps, VLANs, failover)
##############################################################################
### Broadcast domains (MTU 9000 for storage data; 1500 for mgmt/intercluster) ###
network port broadcast-domain create -ipspace Default \
    -broadcast-domain mgmt    -mtu 1500
network port broadcast-domain create -ipspace Default \
    -broadcast-domain nas     -mtu 9000
network port broadcast-domain create -ipspace Default \
    -broadcast-domain san     -mtu 9000
network port broadcast-domain create -ipspace Default \
    -broadcast-domain icl     -mtu 1500

### LACP ifgrps, VLANs, port-to-BC assignment, and failover groups ###
##
### ============================================================================ ###

### CHOOSE ONE OPTION based on what the customer's switching + cabling allows. ###

### Run ONE option block below. Do not mix. ###
##
###   Option A -- ONE ifgrp per node (a0a only) ###

###               Use when the customer only has one set of uplink ports per ###

###               node, or only one LACP group per host is supported by their ###

###               switch fabric. ###
##
###   Option B -- TWO ifgrps per node (a0a + a0b)        [PREFERRED] ###

###               Splits uplinks across two LACP groups for redundancy across ###

###               NIC cards / PCIe risers / leaf switches. Requires twice the ###

###               uplink ports and a matching LACP config on the switch side. ###

### ============================================================================ ###

### ===== Option A: ONE ifgrp per node (a0a) ===== ###
# LACP ifgrp per node -- all uplink ports in a0a
network port ifgrp create -node <NODE1_NAME> -ifgrp a0a -distr-func ip -mode multimode_lacp
network port ifgrp create -node <NODE2_NAME> -ifgrp a0a -distr-func ip -mode multimode_lacp

network port ifgrp add-port -node <NODE1_NAME> -ifgrp a0a -port <PORT_A>
network port ifgrp add-port -node <NODE1_NAME> -ifgrp a0a -port <PORT_B>
network port ifgrp add-port -node <NODE1_NAME> -ifgrp a0a -port <PORT_C>
network port ifgrp add-port -node <NODE1_NAME> -ifgrp a0a -port <PORT_D>
network port ifgrp add-port -node <NODE2_NAME> -ifgrp a0a -port <PORT_A>
network port ifgrp add-port -node <NODE2_NAME> -ifgrp a0a -port <PORT_B>
network port ifgrp add-port -node <NODE2_NAME> -ifgrp a0a -port <PORT_C>
network port ifgrp add-port -node <NODE2_NAME> -ifgrp a0a -port <PORT_D>

# Set MTU on a0a so VLAN children can inherit 9000
network port modify -node <NODE1_NAME> -port a0a -mtu 9000
network port modify -node <NODE2_NAME> -port a0a -mtu 9000

# VLAN interfaces on a0a
vlan create -node <NODE1_NAME> -port a0a -vlan-id <MGMT_VLAN_ID>
vlan create -node <NODE1_NAME> -port a0a -vlan-id <ICL_VLAN_ID>
vlan create -node <NODE1_NAME> -port a0a -vlan-id <NAS_VLAN_ID>
vlan create -node <NODE1_NAME> -port a0a -vlan-id <SAN_VLAN_ID>
vlan create -node <NODE2_NAME> -port a0a -vlan-id <MGMT_VLAN_ID>
vlan create -node <NODE2_NAME> -port a0a -vlan-id <ICL_VLAN_ID>
vlan create -node <NODE2_NAME> -port a0a -vlan-id <NAS_VLAN_ID>
vlan create -node <NODE2_NAME> -port a0a -vlan-id <SAN_VLAN_ID>

# Assign tagged ports to broadcast domains
network port broadcast-domain add-ports -ipspace Default -broadcast-domain mgmt \
    -ports <NODE1_NAME>:a0a-<MGMT_VLAN_ID>,<NODE2_NAME>:a0a-<MGMT_VLAN_ID>
network port broadcast-domain add-ports -ipspace Default -broadcast-domain icl \
    -ports <NODE1_NAME>:a0a-<ICL_VLAN_ID>,<NODE2_NAME>:a0a-<ICL_VLAN_ID>
network port broadcast-domain add-ports -ipspace Default -broadcast-domain nas \
    -ports <NODE1_NAME>:a0a-<NAS_VLAN_ID>,<NODE2_NAME>:a0a-<NAS_VLAN_ID>
network port broadcast-domain add-ports -ipspace Default -broadcast-domain san \
    -ports <NODE1_NAME>:a0a-<SAN_VLAN_ID>,<NODE2_NAME>:a0a-<SAN_VLAN_ID>

# Failover groups (one path per node via a0a)
network interface failover-groups create -vserver <CLUSTER_NAME> -failover-group fg-mgmt \
    -targets <NODE1_NAME>:a0a-<MGMT_VLAN_ID>,<NODE2_NAME>:a0a-<MGMT_VLAN_ID>
network interface failover-groups create -vserver <CLUSTER_NAME> -failover-group fg-icl \
    -targets <NODE1_NAME>:a0a-<ICL_VLAN_ID>,<NODE2_NAME>:a0a-<ICL_VLAN_ID>

### ===== END Option A ===== ###

### ===== Option B: TWO ifgrps per node (a0a + a0b)  [PREFERRED] ===== ###
# LACP ifgrps per node -- uplinks split across a0a and a0b for redundancy
network port ifgrp create -node <NODE1_NAME> -ifgrp a0a -distr-func ip -mode multimode_lacp
network port ifgrp create -node <NODE1_NAME> -ifgrp a0b -distr-func ip -mode multimode_lacp
network port ifgrp create -node <NODE2_NAME> -ifgrp a0a -distr-func ip -mode multimode_lacp
network port ifgrp create -node <NODE2_NAME> -ifgrp a0b -distr-func ip -mode multimode_lacp

network port ifgrp add-port -node <NODE1_NAME> -ifgrp a0a -port <PORT_A>
network port ifgrp add-port -node <NODE1_NAME> -ifgrp a0a -port <PORT_B>
network port ifgrp add-port -node <NODE1_NAME> -ifgrp a0b -port <PORT_C>
network port ifgrp add-port -node <NODE1_NAME> -ifgrp a0b -port <PORT_D>
network port ifgrp add-port -node <NODE2_NAME> -ifgrp a0a -port <PORT_A>
network port ifgrp add-port -node <NODE2_NAME> -ifgrp a0a -port <PORT_B>
network port ifgrp add-port -node <NODE2_NAME> -ifgrp a0b -port <PORT_C>
network port ifgrp add-port -node <NODE2_NAME> -ifgrp a0b -port <PORT_D>

# Set MTU on both ifgrps so VLAN children can inherit 9000
network port modify -node <NODE1_NAME> -port a0a -mtu 9000
network port modify -node <NODE1_NAME> -port a0b -mtu 9000
network port modify -node <NODE2_NAME> -port a0a -mtu 9000
network port modify -node <NODE2_NAME> -port a0b -mtu 9000

# VLAN interfaces on both ifgrps
vlan create -node <NODE1_NAME> -port a0a -vlan-id <MGMT_VLAN_ID>
vlan create -node <NODE1_NAME> -port a0b -vlan-id <MGMT_VLAN_ID>
vlan create -node <NODE2_NAME> -port a0a -vlan-id <MGMT_VLAN_ID>
vlan create -node <NODE2_NAME> -port a0b -vlan-id <MGMT_VLAN_ID>
vlan create -node <NODE1_NAME> -port a0a -vlan-id <NAS_VLAN_ID>
vlan create -node <NODE1_NAME> -port a0b -vlan-id <NAS_VLAN_ID>
vlan create -node <NODE2_NAME> -port a0a -vlan-id <NAS_VLAN_ID>
vlan create -node <NODE2_NAME> -port a0b -vlan-id <NAS_VLAN_ID>
vlan create -node <NODE1_NAME> -port a0a -vlan-id <SAN_VLAN_ID>
vlan create -node <NODE1_NAME> -port a0b -vlan-id <SAN_VLAN_ID>
vlan create -node <NODE2_NAME> -port a0a -vlan-id <SAN_VLAN_ID>
vlan create -node <NODE2_NAME> -port a0b -vlan-id <SAN_VLAN_ID>
vlan create -node <NODE1_NAME> -port a0a -vlan-id <ICL_VLAN_ID>
vlan create -node <NODE1_NAME> -port a0b -vlan-id <ICL_VLAN_ID>
vlan create -node <NODE2_NAME> -port a0a -vlan-id <ICL_VLAN_ID>
vlan create -node <NODE2_NAME> -port a0b -vlan-id <ICL_VLAN_ID>

# Assign tagged ports to broadcast domains -- both ifgrps participate
network port broadcast-domain add-ports -ipspace Default -broadcast-domain mgmt \
    -ports <NODE1_NAME>:a0a-<MGMT_VLAN_ID>,<NODE1_NAME>:a0b-<MGMT_VLAN_ID>,<NODE2_NAME>:a0a-<MGMT_VLAN_ID>,<NODE2_NAME>:a0b-<MGMT_VLAN_ID>
network port broadcast-domain add-ports -ipspace Default -broadcast-domain icl \
    -ports <NODE1_NAME>:a0a-<ICL_VLAN_ID>,<NODE1_NAME>:a0b-<ICL_VLAN_ID>,<NODE2_NAME>:a0a-<ICL_VLAN_ID>,<NODE2_NAME>:a0b-<ICL_VLAN_ID>
network port broadcast-domain add-ports -ipspace Default -broadcast-domain nas \
    -ports <NODE1_NAME>:a0a-<NAS_VLAN_ID>,<NODE1_NAME>:a0b-<NAS_VLAN_ID>,<NODE2_NAME>:a0a-<NAS_VLAN_ID>,<NODE2_NAME>:a0b-<NAS_VLAN_ID>
network port broadcast-domain add-ports -ipspace Default -broadcast-domain san \
    -ports <NODE1_NAME>:a0a-<SAN_VLAN_ID>,<NODE1_NAME>:a0b-<SAN_VLAN_ID>,<NODE2_NAME>:a0a-<SAN_VLAN_ID>,<NODE2_NAME>:a0b-<SAN_VLAN_ID>

# Failover groups (both paths per node)
network interface failover-groups create -vserver <CLUSTER_NAME> -failover-group fg-mgmt \
    -targets <NODE1_NAME>:a0a-<MGMT_VLAN_ID>,<NODE1_NAME>:a0b-<MGMT_VLAN_ID>,<NODE2_NAME>:a0a-<MGMT_VLAN_ID>,<NODE2_NAME>:a0b-<MGMT_VLAN_ID>
network interface failover-groups create -vserver <CLUSTER_NAME> -failover-group fg-icl \

### ===== END Option B ===== ###

### Phase 5 -- Storage layout (aggregates + HA)
##############################################################################
### Data aggregates -- auto-provision ###
# auto-provision picks the right RAID type, disk count, and spare allocation
# for the platform automatically. Preview first (read-only), then apply.
storage aggregate auto-provision -verbose
storage aggregate auto-provision


### Rename root aggregates (standardize to n<N>_aggr0 -- matches the LIF naming convention) ###
# Must happen BEFORE Phase 6 SVM creation, since SVMs reference aggregates by name.
storage aggregate rename -aggregate aggr0_<NODE1_NAME>_0 -newname n1_aggr0
storage aggregate rename -aggregate aggr0_<NODE2_NAME>_0 -newname n2_aggr0


### Enable HA and storage failover ###
cluster ha modify -configured true
storage failover modify -node * -enabled true
storage failover modify -node * -auto-giveback true
storage failover modify -node * -onreboot true

# VERIFY:
storage failover show
cluster ha show


### Phase 6 -- SVM: NAS (<NAS_SVM>) -- NFS + SMB/CIFS
##############################################################################
vserver create -vserver <NAS_SVM> -subtype default -rootvolume <NAS_SVM>_root \
    -rootvolume-security-style mixed -language C.UTF-8 \
    -snapshot-policy default -aggregate <NODE1_NAME>_aggr1
# NAS root vol uses 'mixed' security style so NFS clients see UNIX perms and
# CIFS clients see NTFS perms on the same SVM. (SAN SVM stays 'unix'.)
vserver modify -vserver <NAS_SVM> -allowed-protocols nfs,cifs

### Load-sharing mirrors for root volume on each node ###
volume create -vserver <NAS_SVM> -volume <NAS_SVM>_root_m1 -aggregate <NODE1_NAME>_aggr1 \
    -size 1g -type DP
volume create -vserver <NAS_SVM> -volume <NAS_SVM>_root_m2 -aggregate <NODE2_NAME>_aggr1 \
    -size 1g -type DP
snapmirror create -source-path <CLUSTER_NAME>://<NAS_SVM>/<NAS_SVM>_root \
    -destination-path <CLUSTER_NAME>://<NAS_SVM>/<NAS_SVM>_root_m1 -type LS \
    -schedule hourly
snapmirror create -source-path <CLUSTER_NAME>://<NAS_SVM>/<NAS_SVM>_root \
    -destination-path <CLUSTER_NAME>://<NAS_SVM>/<NAS_SVM>_root_m2 -type LS \
    -schedule hourly
snapmirror initialize-ls-set -source-path <CLUSTER_NAME>://<NAS_SVM>/<NAS_SVM>_root

### DNS on the SVM ###
vserver services name-service dns create -vserver <NAS_SVM> \
    -domains <AD_DOMAIN> -name-servers <DNS_SERVER_1>,<DNS_SERVER_2> -state enabled

### NFS server (v3 + v4.1) + vstorage for VAAI ###
vserver nfs create -vserver <NAS_SVM> -access true -v3 enabled -v4.1 enabled \
    -vstorage enabled -v4-id-domain <AD_DOMAIN>
# Idempotent re-assert of vstorage (required for VMware VAAI NFS Plug-in)
vserver nfs modify -vserver <NAS_SVM> -vstorage enabled
vserver nfs modify -vserver <NAS_SVM> -tcp-max-xfer-size 1048576

### Tightened default export policy ###
# The 'default' policy starts with NO rules -- LEAVE IT THAT WAY.
# Create per-workload export policies and rules. Example:
vserver export-policy create -vserver <NAS_SVM> -policyname vmware_nfs
vserver export-policy rule create -vserver <NAS_SVM> -policyname vmware_nfs \
    -clientmatch <CIDR_CLIENTS> -rorule sys -rwrule sys -superuser sys \
    -anon 65534 -protocol nfs3,nfs41 -ruleindex 1

### NAS data LIFs + default route -- REQUIRED before CIFS AD join ###
# The CIFS create below performs an AD domain join at execution time. The SVM
# must already have at least one data LIF up AND a route out, or the join will
# fail (cannot reach DNS / cannot reach a domain controller).

# NFS LIFs (one per node)
network interface create -vserver <NAS_SVM> -lif n1_nfs_lif01 \
    -service-policy default-data-files -address <NAS_LIF_1_IP> \
    -netmask <NAS_MASK> -home-node <NODE1_NAME> \
    -home-port a0a-<NAS_VLAN_ID> -status-admin up \
    -failover-policy broadcast-domain-wide -auto-revert true
network interface create -vserver <NAS_SVM> -lif n2_nfs_lif02 \
    -service-policy default-data-files -address <NAS_LIF_2_IP> \
    -netmask <NAS_MASK> -home-node <NODE2_NAME> \
    -home-port a0a-<NAS_VLAN_ID> -status-admin up \
    -failover-policy broadcast-domain-wide -auto-revert true

# CIFS LIFs (one per node; same service-policy on ONTAP 9.13+)
network interface create -vserver <NAS_SVM> -lif n1_cifs_lif01 \
    -service-policy default-data-files -address <SMB_LIF_1_IP> \
    -netmask <NAS_MASK> -home-node <NODE1_NAME> \
    -home-port a0b-<NAS_VLAN_ID> -status-admin up \
    -failover-policy broadcast-domain-wide -auto-revert true
network interface create -vserver <NAS_SVM> -lif n2_cifs_lif02 \

    -home-port a0b-<NAS_VLAN_ID> -status-admin up \

### CIFS server with AD join + SMB hardening ###

    -is-signing-required true -is-smb-encryption-required true \
    -kerberos-clock-skew 5 -is-aes-encryption-enabled true \
    -lm-compatibility-level krb

    -is-advertise-dfs-enabled false -is-shadowcopy-enabled true \
    -smb1-enabled-for-dc-connections false
# Share defaults: enable Access-Based Enumeration globally on new shares

    -share-properties browsable    2>/dev/null || true

vserver cifs show -vserver <NAS_SVM>
